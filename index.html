function startRecognition() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert("このブラウザは音声認識に対応していません。Chrome推奨です。"); return; } const recognition = new SpeechRecognition(); recognition.lang = "en-US"; recognition.interimResults = false; recognition.continuous = false; recognition.maxAlternatives = 1; recognition.onstart = function() { console.log("音声認識開始"); }; recognition.onresult = function(event) { const spokenText = event.results[0][0].transcript; document.getElementById("recognizedText").innerText = spokenText; const targetText = document.getElementById("targetText").value; const similarity = calculateSimilarity(targetText, spokenText); document.getElementById("score").innerText = similarity; }; recognition.onerror = function(event) { console.log(event.error); if (event.error === "aborted") { document.getElementById("recognizedText").innerText = "音声認識が中断されました。もう一度試してください。"; return; } alert("エラー: " + event.error); }; recognition.onend = function() { console.log("音声認識終了"); }; try { recognition.start(); } catch (e) { console.log(e); } }
